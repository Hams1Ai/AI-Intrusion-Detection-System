def predict_with_rl(features):
    # features: 1D array or list of numeric values

    x = np.array(features, dtype=float).reshape(1, -1)
    x_scaled = scaler.transform(x)

    # XGBoost probability of attack
    score = float(xgb.predict_proba(x_scaled)[0, 1])  # 0..1
    label = int(xgb.predict(x_scaled)[0])             # 0 or 1

    # state for RL = [risk_score]
    state = np.array([score], dtype=np.float32)
    action, _ = model_rl.predict(state, deterministic=True)
    action = int(action)   # 0 = Ignore, 1 = Block

    return score, label, action
